        if entry.name not in ignore_list and not entry.name.startswith('.'):
            dir_list.append(entry.name)
    return json.dumps(dir_list, ensure_ascii=False)

def copy_file(src, target, file_name, new_name=None, cover=True, suffix=None):
    """
    复制文件的实际执行函数
    参数和你的 inputSchema 完全对应
    """
    # 1. 构造源文件完整路径
    src_path = os.path.join(src, file_name)

    # 2. 确定目标文件名
    if new_name:
        final_name = new_name
    else:
        final_name = os.path.splitext(file_name)[0]  # 去掉原后缀的文件名

    # 3. 如果指定了新后缀，替换掉
    if suffix:
        final_name = final_name + suffix
    else:
        final_name = final_name + os.path.splitext(file_name)[1]  # 保留原后缀

    # 4. 构造目标完整路径
    dst_path = os.path.join(target, final_name)

    # 5. 检查是否覆盖
    if os.path.exists(dst_path) and not cover:
        return {"success": False, "message": f"文件 {dst_path} 已存在，且 cover=False，操作取消"}

    # 6. 执行复制
    try:
        shutil.copy2(src_path, dst_path)  # copy2 会保留元数据
        return {
            "success": True,
            "message": f"成功将 {src_path} 复制到 {dst_path}",
            "src_path": src_path,
            "dst_path": dst_path
        }
    except FileNotFoundError:
        return {"success": False, "message": f"源文件不存在: {src_path}"}
    except Exception as e:
        return {"success": False, "message": f"复制失败: {str(e)}"}

def build_dynamic_system_prompt():
    """根据注册表动态生成系统提示词（过滤掉不可序列化的函数对象）"""

    clean_tools_desc = {}

def process_user_request(user_input: str):
    """
    【通用 Agent 调度器】
    不绑定任何具体业务，仅负责：接收指令 -> 调用大模型 -> 执行工具 -> 返回结果
    """
    print("[系统] 正在分析您的指令...")

    # 1. 动态生成包含所有工具的 Prompt
    system_prompt = build_dynamic_system_prompt()

    # 2. 将用户指令发给大模型
    ai_response_raw = get_ollama_reply(user_input, system_prompt=system_prompt)
    ai_response = parse_ai_json(ai_response_raw["message"]["content"])

    if not ai_response:
        return {"success": False, "message": "AI 未能返回有效的指令"}

    # 3. 提取工具名和参数
    tool_name = ai_response.get("tool_name")
    arguments = ai_response.get("arguments", {})

    # 4. 校验并执行工具
    if tool_name not in TOOL_REGISTRY:
        return {"success": False, "message": f"AI 调用了不存在的工具: {tool_name}"}

    target_func = TOOL_REGISTRY[tool_name]["function"]
    print(f"[系统] AI 决定调用工具: {tool_name}，参数: {arguments}")

    try:
        result = target_func(**arguments)
        return result
    except Exception as e:
        return {"success": False, "message": f"执行异常: {str(e)}"}
